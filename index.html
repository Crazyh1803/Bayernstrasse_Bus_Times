async function fetchStopData(stationId, allowedLines) { // Querying using the stable v6 gateway structure const url = `https://v6.db.transport.rest/stops/${stationId}/departures?duration=60&results=20`; logDebug(`Polling station ID: ${stationId}...`); try { const response = await fetch(url); if (!response.ok) { logDebug(`HTTP Response Error: ${response.status}`); return getSimulationData(stationId, allowedLines); } const data = await response.json(); // Handle differences between different API wrapper arrays const departures = data.departures || data; if (!departures || departures.length === 0) { logDebug(`Zero active departures returned for ${stationId}`); return getSimulationData(stationId, allowedLines); } logDebug(`Successfully connected! Processing raw transit matrix...`); return departures .filter(dep => { // MODERN API CORRECTION: Safely extract line markers across varying sub-objects let lineName = ""; if (dep.line) { lineName = dep.line.name || dep.line.id || ""; } // Cross-check if the line number matches our active filter (18, 21, or 22) const matchesLine = allowedLines.some(l => lineName.includes(l)); if (!matchesLine) return false; // Separate out any outbound commuter traffic heading to distant suburbs const dest = dep.direction ? dep.direction.toLowerCase() : ""; const isOutbound = dest.includes("medenbach") || dest.includes("naurod") || dest.includes("eigenheim") || dest.includes("bierstadt ost"); return !isOutbound; }) .map(dep => { // Extract the clean numeric representation of the line let cleanLineNum = "??"; if (dep.line && dep.line.name) { cleanLineNum = dep.line.name.replace(/\D/g, ''); } else if (dep.line && dep.line.id) { cleanLineNum = dep.line.id.replace(/\D/g, ''); } if (!cleanLineNum || cleanLineNum === "") { cleanLineNum = allowedLines[0]; // Fallback string assignment if stripped completely } return { line: cleanLineNum, destination: dep.direction || "Wiesbaden Center", origin: stationId === STOP_KAPPENBERGWEG ? "Kappenbergweg" : "Württembergstr.", targetTime: dep.when || dep.plannedWhen }; }); } catch (error) { logDebug(`Parser exception hit: ${error.message}. Initializing safe simulation fallback.`); return getSimulationData(stationId, allowedLines); } }
